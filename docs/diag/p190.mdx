};

export const sidebar = (ch, t) => {
  const g = GROUPS.map((x) => ({ ...x, items: x.items.map((it) => ({ ...it })) }));
  const find = (title) => g.flatMap((x) => x.items).find((it) => it.t === title);
  if (ch === 0) find("Refactor session auth").s = t < 3.1 ? "working" : "waiting";
  // A split stays in the same tab; the tab picks up the agent's status.
  if (ch === 1 && t >= 3.2) find("~/code/orbit").s = t < 6.4 ? "working" : "done";
  if ((ch === 3 && t >= 4.1) || ch === 4) {
    find("~/code/orbit").on = false;
    g[0].items.unshift({ a: "claude", t: "Handle empty daily summaries", s: ch === 4 && t >= 5.2 ? "done" : "working", on: true });
  }
  return g;
};
